<script>
  (function() {
    const mobileNavToggle = document.querySelector('.mobile-nav-toggle');
    const mobileNav = document.querySelector('.mobile-nav');
    const mobileNavClose = document.querySelector('.mobile-nav-close');
    const mobileNavOverlay = document.querySelector('.mobile-nav-overlay');
    const accordions = document.querySelectorAll('.mobile-nav-accordion');

    function openMobileNav() {
      mobileNav.classList.add('active');
      mobileNavToggle.setAttribute('aria-expanded', 'true');
      document.body.style.overflow = 'hidden';
    }

    function closeMobileNav() {
      mobileNav.classList.remove('active');
      mobileNavToggle.setAttribute('aria-expanded', 'false');
      document.body.style.overflow = '';
    }

    // Artist Dropdown
    const artistDropdown = document.querySelector('.artist-dropdown');
    const artistButton = document.querySelector('.artist-link-subtle');
    
    if (artistButton) {
      artistButton.addEventListener('click', function(e) {
        e.stopPropagation();
        const isOpen = artistDropdown.classList.contains('is-open');
        artistDropdown.classList.toggle('is-open');
        artistButton.setAttribute('aria-expanded', !isOpen);
      });
      
      // Close on click outside
      document.addEventListener('click', function(e) {
        if (!artistDropdown.contains(e.target)) {
          artistDropdown.classList.remove('is-open');
          artistButton.setAttribute('aria-expanded', 'false');
        }
      });
      
      // Close on Escape key
      document.addEventListener('keydown', function(e) {
        if (e.key === 'Escape' && artistDropdown.classList.contains('is-open')) {
          artistDropdown.classList.remove('is-open');
          artistButton.setAttribute('aria-expanded', 'false');
          artistButton.focus();
        }
      });
    }
    
    if (mobileNavToggle) {
      mobileNavToggle.addEventListener('click', function() {
        if (mobileNav.classList.contains('active')) {
          closeMobileNav();
        } else {
          openMobileNav();
        }
      });
    }

    if (mobileNavClose) {
      mobileNavClose.addEventListener('click', closeMobileNav);
    }

    if (mobileNavOverlay) {
      mobileNavOverlay.addEventListener('click', closeMobileNav);
    }

    accordions.forEach(function(accordion) {
      accordion.addEventListener('click', function() {
        const isExpanded = this.getAttribute('aria-expanded') === 'true';
        this.setAttribute('aria-expanded', !isExpanded);
      });
    });
  })();
</script>

<script>
// Premium option selector interactions
document.addEventListener('DOMContentLoaded', function() {
  // Size selection
  const sizeOptions = document.querySelectorAll('[data-option-type="size"]');
  sizeOptions.forEach(option => {
    option.addEventListener('click', function() {
      sizeOptions.forEach(opt => opt.classList.remove('selected'));
      this.classList.add('selected');
      const radio = this.querySelector('input[type="radio"]');
      if (radio) radio.checked = true;
      updateVariant();
    });
  });

  // Finish selection
  const finishOptions = document.querySelectorAll('[data-option-type="finish"]');
  finishOptions.forEach(option => {
    option.addEventListener('click', function() {
      finishOptions.forEach(opt => opt.classList.remove('selected'));
      this.classList.add('selected');
      const radio = this.querySelector('input[type="radio"]');
      if (radio) radio.checked = true;
      updateVariant();
    });
  });

  // Frame selection
  const frameToggles = document.querySelectorAll('.frame-toggle');
  frameToggles.forEach(toggle => {
    toggle.addEventListener('click', function() {
      frameToggles.forEach(t => t.classList.remove('selected'));
      this.classList.add('selected');
      const radio = this.querySelector('input[type="radio"]');
      if (radio) radio.checked = true;
      updatePrice();
    });
  });

  // Set initial selected states
  sizeOptions[0]?.classList.add('selected');
  finishOptions[0]?.classList.add('selected');
  document.getElementById('no-frame-option')?.classList.add('selected');
  
  // Initialize frame availability based on default finish (Paper)
  const defaultFinish = document.querySelector('input[name="finish"]:checked')?.value || 'Paper';
  updateFrameAvailability(defaultFinish);

  // Size guide toggle
  const sizeGuideToggle = document.getElementById('size-guide-toggle');
  const sizeGuideContent = document.getElementById('size-guide-content');
  
  if (sizeGuideToggle && sizeGuideContent) {
    sizeGuideToggle.addEventListener('click', function() {
      const isOpen = sizeGuideContent.classList.contains('open');
      if (isOpen) {
        sizeGuideContent.classList.remove('open');
        sizeGuideToggle.classList.remove('open');
      } else {
        sizeGuideContent.classList.add('open');
        sizeGuideToggle.classList.add('open');
      }
    });
  }
});

function updateVariant() {
  const size = document.querySelector('input[name="size"]:checked')?.value;
  const baseFinish = document.querySelector('input[name="finish"]:checked')?.value;
  const frame = document.querySelector('input[name="frame"]:checked')?.value;
  
  // Map Paper + Frame to Framed-[Color] variants
  let actualFinish = baseFinish;
  if (baseFinish === 'Paper' && frame && frame !== 'none') {
    actualFinish = `Framed-${frame.charAt(0).toUpperCase() + frame.slice(1)}`;
  }
  
  console.log('Variant updated:', { size, baseFinish, frame, actualFinish });
  
  // Dispatch custom event with mapped finish
  document.dispatchEvent(new CustomEvent('variantChange', { 
    detail: { size, finish: actualFinish }
  }));
  
  // Update frame options availability
  updateFrameAvailability(baseFinish);
}

function updatePrice() {
  const frame = document.querySelector('input[name="frame"]:checked')?.value;
  console.log('Frame option:', frame);
