  (function() {
    const mobileNavToggle = document.querySelector('.mobile-nav-toggle');
    const mobileNav = document.querySelector('.mobile-nav');
    const mobileNavClose = document.querySelector('.mobile-nav-close');
    const mobileNavOverlay = document.querySelector('.mobile-nav-overlay');
    const accordions = document.querySelectorAll('.mobile-nav-accordion');

    function openMobileNav() {
      mobileNav.classList.add('active');
      mobileNavToggle.setAttribute('aria-expanded', 'true');
      document.body.style.overflow = 'hidden';
    }

    function closeMobileNav() {
      mobileNav.classList.remove('active');
      mobileNavToggle.setAttribute('aria-expanded', 'false');
      document.body.style.overflow = '';
    }

    // Artist Dropdown
    const artistDropdown = document.querySelector('.artist-dropdown');
    const artistButton = document.querySelector('.artist-link-subtle');
    
    if (artistButton) {
      artistButton.addEventListener('click', function(e) {
        e.stopPropagation();
        const isOpen = artistDropdown.classList.contains('is-open');
        artistDropdown.classList.toggle('is-open');
        artistButton.setAttribute('aria-expanded', !isOpen);
      });
      
      // Close on click outside
      document.addEventListener('click', function(e) {
        if (!artistDropdown.contains(e.target)) {
          artistDropdown.classList.remove('is-open');
          artistButton.setAttribute('aria-expanded', 'false');
        }
      });
      
      // Close on Escape key
      document.addEventListener('keydown', function(e) {
        if (e.key === 'Escape' && artistDropdown.classList.contains('is-open')) {
          artistDropdown.classList.remove('is-open');
          artistButton.setAttribute('aria-expanded', 'false');
          artistButton.focus();
        }
      });
    }
    
    if (mobileNavToggle) {
      mobileNavToggle.addEventListener('click', function() {
        if (mobileNav.classList.contains('active')) {
          closeMobileNav();
        } else {
          openMobileNav();
        }
      });
    }

    if (mobileNavClose) {
      mobileNavClose.addEventListener('click', closeMobileNav);
    }

    if (mobileNavOverlay) {
      mobileNavOverlay.addEventListener('click', closeMobileNav);
    }

    accordions.forEach(function(accordion) {
      accordion.addEventListener('click', function() {
        const isExpanded = this.getAttribute('aria-expanded') === 'true';
        this.setAttribute('aria-expanded', !isExpanded);
      });
    });
  })();
</script>
